class Solution {
    public boolean isPalindrome(ListNode head) {
        ListNode slow = head, fast = head, prev = null;
        while (fast != null && fast.next != null) {
            fast = fast.next.next;
            ListNode next = slow.next;
            slow.next = prev;
            prev = slow;
            slow = next;
        }
        if (fast != null) slow = slow.next; 
        while (prev != null && prev.val == slow.val) {
            prev = prev.next;
            slow = slow.next;
        }
        return prev == null;
    }
}



Input: head = [1,2,2,1]
Output: true
