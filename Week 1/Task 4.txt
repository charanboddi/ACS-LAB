class Solution {
    public int[] getSumAbsoluteDifferences(int[] nums) {
        int n = nums.length;
        int[] result = new int[n];

              int totalSum = 0;
        for (int num : nums) {
            totalSum += num;
        }

        int leftSum = 0;

        for (int i = 0; i < n; i++) {
            int current = nums[i];

                 int left = current * i - leftSum;

                   int right = (totalSum - leftSum - current)
                        - current * (n - i - 1);

            result[i] = left + right;

            leftSum += current;
        }

        return result;
    }
}


Input: nums = [2,3,5]
Output: [4,3,5]
